public class Deserialization {
    public static class StateDeserializer implements JsonDeserializer<State> {

        @Override
        public State deserialize(JsonElement json, Type typeOfT, JsonDeserializationContext context) throws JsonParseException {
            State state = new State();
            JsonObject object = json.getAsJsonObject();
            JsonObject usersObj = object.getAsJsonObject("users");
            for (Map.Entry<String, JsonElement> entry : usersObj.entrySet()) {
                JsonElement userElement = entry.getValue();
                String role = userElement.getAsJsonObject().get("role").getAsString();
                Class<? extends User> type;
                switch (role) {
                    case "Admin":
                        type = Admin.class;
                        break;
                    case "Driver":
                        type = Driver.class;
                        break;
                    default:
                        type = Customer.class;
                        break;
                }

                User user = context.deserialize(userElement, type);

                state.users.put(user.getId(), user);
            }

            JsonElement menuElement = object.get("menu");
            Menu menu = context.deserialize(menuElement, Menu.class);
            state.menu = menu;

            JsonArray queue = object.getAsJsonArray("orderQueue");

            for (JsonElement element : queue) {
                Order order = context.deserialize(element, Order.class);
            if (order != null) state.orderQueue.add(order);
            }
            return state;
        }
    }
    public static class OrderDeserializer implements JsonDeserializer<Order> {

        @Override
        public Order deserialize(JsonElement json, Type typeOfT, JsonDeserializationContext context) throws JsonParseException {
            JsonObject object = json.getAsJsonObject();
            UUID id = UUID.fromString(object.get("id").getAsString());
            UUID customerId = UUID.fromString(object.get("customerId").getAsString());
            Order o = new Order(id, customerId);
            JsonArray items = object.getAsJsonArray("items");
            for (JsonElement element : items) {
                QuantizedItem item = context.deserialize(element, QuantizedItem.class);
                o.addQuantizedItem(item);
            }
            return o;
        }
    }
    public static class QuantizedItemDeserializer implements JsonDeserializer<QuantizedItem> {

        @Override
        public QuantizedItem deserialize(JsonElement json, Type typeOfT, JsonDeserializationContext context) throws JsonParseException {
            JsonObject object = json.getAsJsonObject();
            return new QuantizedItem(UUID.fromString(object.get("itemId").getAsString()),object.get("quantity").getAsInt());
        }
    }
    public static class ItemDeserializer implements JsonDeserializer<Item> {
        @Override
        public Item deserialize(JsonElement json, Type typeOfT, JsonDeserializationContext context) throws JsonParseException {
            JsonObject object = json.getAsJsonObject();
            String name = object.get("name").getAsString();
            String description = object.get("description").getAsString();
            Double price = object.get("price").getAsDouble();
            UUID id = UUID.fromString(object.get("id").getAsString());
            return new Item(name, description, price, id);
        }
    }
    public static class AdminDeserializer implements JsonDeserializer<Admin> {

        @Override
        public Admin deserialize(JsonElement json, Type typeOfT, JsonDeserializationContext context) throws JsonParseException {
            JsonObject object = json.getAsJsonObject();
            return new Admin(object.get("name").getAsString(),UUID.fromString(object.get("id").getAsString()),hexToBytes(object.get("phash").getAsString()));
        }
    }
    public static class CustomerDeserializer implements JsonDeserializer<Customer> {

        @Override
        public Customer deserialize(JsonElement json, Type typeOfT, JsonDeserializationContext context) throws JsonParseException {
            JsonObject object = json.getAsJsonObject();
            return new Customer(object.get("name").getAsString(),UUID.fromString(object.get("id").getAsString()),hexToBytes(object.get("phash").getAsString()));
        }
    }
    public static class DriverDeserializer implements JsonDeserializer<Driver> {

        @Override
        public Driver deserialize(JsonElement json, Type typeOfT, JsonDeserializationContext context) throws JsonParseException {
            JsonObject object = json.getAsJsonObject();
            Driver driver = new Driver(object.get("name").getAsString(),UUID.fromString(object.get("id").getAsString()),hexToBytes(object.get("phash").getAsString()));
            Rating[] ratings = context.deserialize(object.get("Rating"), Rating[].class);
            for (Rating r : ratings) {
                driver.addRate(r);
            }
            return driver;
        }
    }
    public static class UserRatingDeserializer implements JsonDeserializer<Rating> {

        @Override
        public Rating deserialize(JsonElement json, Type typeOfT, JsonDeserializationContext context) throws JsonParseException {
            JsonObject object = json.getAsJsonObject();
            return new Rating(UUID.fromString(object.get("creatorId").getAsString()),object.get("score").getAsInt(),object.get("timestamp").getAsString());
        }
    }
    private static byte[] hexToBytes(String hex) {
        int len = hex.length();
        if (len % 2 != 0) {
            throw new IllegalArgumentException("Hex string must have even length");
        }

        byte[] result = new byte[len / 2];
        for (int i = 0; i < len; i += 2) {
            int high = Character.digit(hex.charAt(i), 16);
            int low  = Character.digit(hex.charAt(i + 1), 16);

            if (high == -1 || low == -1) {
                throw new IllegalArgumentException("Invalid hex character");
            }

            result[i / 2] = (byte) ((high << 4) + low);
        }
        return result;
    }

public class Driver extends User {

    private final Queue<Rating> ratings;
    
    protected Driver(String name, UUID id, String password){
        super(name, id, password);
        this.status = Status.FREE;
        orders = new ArrayList<>();
        ratings = new LinkedList<>();
    }
    protected Driver(User u) { 
        super(u.getName(), u.getId(), u.getPasswordHash()); 
        orders = new ArrayList<>();
        ratings = new LinkedList<>();
        }

    protected Driver(String name, UUID id, byte[] phashes) {
        super(name,id,phashes);
        this.status = Status.FREE;
        orders = new ArrayList<>();
        ratings = new LinkedList<>();
    }
    public Driver(String name, String password) {
        this(name, UUID.randomUUID(), password);
    }
    public void addOrder(Order order) {
        orders.add(order);
    }
    public ArrayList<Order> getOrders() {
        return orders;
    }
    public Status getStatus() {
        return status;
    }
    public void goOffline() {
        status = Status.OFFLINE;
    }
    public void goOnline() {
        status = Status.FREE;
    }
    public double averageRate() {
        double avg = 0;
        for (Rating rate : ratings) {
            avg += rate.getScore();
        }
        return avg / ratings.size();
    }
    public void addRate(Rating rate) {
        if (ratings.size() > 5) {
            ratings.poll();
            ratings.add(rate);
        }
        ratings.add(rate);
    }
    public Queue<Rating> getRates() {
        return ratings;
    }
    @Override
    public String getRole() {
        return "Driver";
    }
    public static enum Status {
        FREE,
        BUSY,
        OFFLINE
    }

public class Order {

    private Customer customer;
    private UUID customerId;
    private final ArrayList<QuantizedItem> items;
    private Status status;
    private final UUID id;

    public Order(Customer customer) {
        this.customer = customer;
        this.customerId = customer.getId();
        items = new ArrayList<>();
        this.id = UUID.randomUUID();
        this.status = Status.PLACED;
    }

    protected Order(UUID id, UUID customerId) {
        this.customerId = customerId;
        items = new ArrayList<>();
        this.id = id;
        this.status = Status.PLACED;
    }

    public void addItem(Item item, int quantity) {
        items.add(new QuantizedItem(item, quantity));
    }

    public void addQuantizedItem(QuantizedItem item) {
        items.add(item);
    }

    public void removeItem(Item item) {
        for (QuantizedItem quantizedItem : items) {
            if (quantizedItem.getItemId().equals(item.getId())) {
                items.remove(quantizedItem);
                return;
            }
        }
    }

    public Customer getCustomer() {
        return customer;
    }

    public ArrayList<QuantizedItem> getItems() {
        return items;
    }

    public Status getStatus() {
        return status;
    }

    public void updateStatus(Status status) {
        this.status = status;
    }

    public UUID getId() {
        return this.id;
    }

    public static enum Status {
        PLACED,
        ACCEPTED,
        INPROGRESS,
        DELIVERED
    }

    protected void setCustomer(Customer customer) {
        this.customer = customer;
        this.customerId = customer.getId();
    }

    protected UUID getCustomerId() {
        return customerId;
    }

    @Override
    public int hashCode() {
        return id.hashCode();
    }
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Order)) {
            return false;
        }
        Order other = (Order) o;
        return this.id.equals(other.id);
    }

import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.Arrays;
import java.util.UUID;

public abstract class User {
    private String name;
    private final UUID id;
    private final byte[] passwordHash;

    public User(String name, String password) {
        this(name, UUID.randomUUID(), password);
    }

    protected User(String name, UUID id, String password) {
        this.name = name;
        this.id = id;
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            passwordHash = digest.digest(password.getBytes());
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException(e);  // Lets try and remove this if possible
        }
    }

    protected User(String name, UUID id, byte[] passwordHash) {
        this.name = name;
        this.id = id;
        this.passwordHash = passwordHash;
    }

    protected byte[] getPasswordHash() {
        return Arrays.copyOf(passwordHash, passwordHash.length);
    }

    protected String getPasswordHashHex() {
        StringBuilder hexBuilder = new StringBuilder(passwordHash.length * 2);
        for (byte b : passwordHash)
            hexBuilder.append(String.format("%02x", b));
        return hexBuilder.toString();
    }

    public boolean checkPassword(String inputPassword) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            return Arrays.equals(digest.digest(inputPassword.getBytes()), passwordHash);
        } catch (NoSuchAlgorithmException e) {
            return false;
        }
    }

    public String getName() {
        return name;
    }

    public UUID getId() {
        return id;
    }

    public abstract String getRole();

    @Override
    public int hashCode() {
        return id.hashCode();
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (!(obj instanceof User))
            return false;
        User other = (User) obj;
        return id.equals(other.id);
    }

public class Restaurant {
    private final State state;

    public Restaurant(State state) {
        this.state = state;
    }

    public void addUser(User u) {
        if (!state.users.containsKey(u.getId()) && !isNameTaken(u.getName(), state.users.values())
                && !isIdTaken(u.getId(), state.users.values())) {
            state.users.put(u.getId(), u);
        }
    }

    public boolean containsUser(User u) {
        return state.users.containsValue(u);
    }

    public boolean containsUser(String name) {
        for (User u : state.users.values()) {
            if (u.getName().equals(name))
                return true;
        }
        return false;
    }

    public void removeUser(User u) {
        state.users.remove(u.getId());
    }

    public User getUser(String name) {
        for (User u : state.users.values()) {
            if (u.getName().equals(name))
                return u;
        }
        return null;
    }

    public Collection<User> getUsers() {
        return Collections.unmodifiableCollection(state.users.values());
    }

    public Admin getAdmin(String name) {
        for (User u : state.users.values()) {
            if (u instanceof Admin && u.getName().equals(name))
                return (Admin) u;
        }
        return null;
    }

    public List<Driver> getDrivers() {
        List<Driver> drivers = new ArrayList<>();
        for (User u : state.users.values()) {
            if (u instanceof Driver)
                drivers.add((Driver) u);
        }
        return drivers;
    }

    public Driver getDriver(String name) {
        for (User u : state.users.values()) {
            if (u instanceof Driver && u.getName().equals(name))
                return (Driver) u;
        }
        return null;
    }

    public Customer getCustomer(String name) {
        for (User u : state.users.values()) {
            if (u instanceof Customer && u.getName().equals(name))
                return (Customer) u;
        }
        return null;
    }

    public static boolean isNameTaken(String name, Collection<? extends User> users) {
        for (User u : users) {
            if (u.getName().equals(name)) {
                return true;
            }
        }
        return false;
    }

    public static boolean isIdTaken(UUID id, Collection<? extends User> users) {
        for (User u : users) {
            if (u.getId().equals(id)) {
                return true;
            }
        }
        return false;
    }

    public Menu getMenu() {
        return state.menu;
    }

    public void addOrder(Order o) {
        state.orderQueue.add(o);
        update();
    }

    public Order nextOrder() {
        return state.orderQueue.poll();
    }

    public void update() {
        assignDriver();
    }
    
    public void writeOrder(Order o) {
        File file = new File("order.txt");
        Gson gson = new GsonBuilder()
                .setPrettyPrinting()
                .registerTypeAdapter(Order.class, new Serialization.OrderSerializer())
                .registerTypeAdapter(Order.class, new Deserialization.OrderDeserializer())
                .registerTypeAdapter(QuantizedItem.class, new Serialization.QuantizedItemSerializer())
                .registerTypeAdapter(QuantizedItem.class, new Deserialization.QuantizedItemDeserializer())
                .create();
        try {
            FileWriter writer = new FileWriter(file);
            gson.toJson(o, writer);
        } catch (IOException e) {
            System.err.println("Failed to save!"); // is this the best way to do this? maybe make a new screen instead
        }
    }

    public Collection<Order> getOrderQueue() {
        return Collections.unmodifiableCollection(state.orderQueue);
    }

    public void addRating(Order order, Rating rate) {
        for (Driver d : getDrivers()) {
            for (Order o : d.getOrders()) {
                if (order.equals(o)) {
                    for (Rating r : d.getRates()) {
                        if (r.equals(rate)) {
                            return;
                        }
                    }
                    d.addRate(rate);
                }
            }
        }
    }

    public void assignDriver() {
        Order order = nextOrder();
        if (order == null)
            return;
        Driver driver = findDriver();
        order.updateStatus(Order.Status.ACCEPTED);
        driver.addOrder(order);
    }

    public Driver findDriver() {
        Driver best = null;
        for (Driver driver : getDrivers()) {
            if (driver.getStatus() == Driver.Status.FREE) {
                if (best == null || driver.averageRate() > best.averageRate()) {
                    best = driver;
                }
            }
        }
        return best;
    }

    public int checkLogin(String user, String pass) {
        if (containsUser(user)) {
            User u = getUser(user);
            if (u.checkPassword(pass)) {
                if (u.getRole().equals("Admin"))
                    return 0;
                if (u.getRole().equals("Driver"))
                    return 1;
                else
                    return 2;
            } else
                return 3;
        } else
            return 3;
    }

public class StateLoader {
    private static Gson gson;

    private static void initGsonIfNull() {
        if (gson != null)
            return;
        gson = new GsonBuilder()
                .setPrettyPrinting()
                .registerTypeAdapter(State.class, new Serialization.StateSerializer())
                .registerTypeAdapter(State.class, new Deserialization.StateDeserializer())
                .registerTypeAdapter(Order.class, new Serialization.OrderSerializer())
                .registerTypeAdapter(Order.class, new Deserialization.OrderDeserializer())
                .registerTypeAdapter(QuantizedItem.class, new Serialization.QuantizedItemSerializer())
                .registerTypeAdapter(QuantizedItem.class, new Deserialization.QuantizedItemDeserializer())
                .registerTypeAdapter(Item.class, new Serialization.ItemSerializer())
                .registerTypeAdapter(Item.class, new Deserialization.ItemDeserializer())
                .registerTypeAdapter(Admin.class, new Serialization.AdminSerializer())
                .registerTypeAdapter(Admin.class, new Deserialization.AdminDeserializer())
                .registerTypeAdapter(Customer.class, new Serialization.CustomerSerializer())
                .registerTypeAdapter(Customer.class, new Deserialization.CustomerDeserializer())
                .registerTypeAdapter(Driver.class, new Serialization.DriverSerializer())
                .registerTypeAdapter(Driver.class, new Deserialization.DriverDeserializer())
                .registerTypeAdapter(Rating.class, new Serialization.UserRatingSerializer())
                .registerTypeAdapter(Rating.class, new Deserialization.UserRatingDeserializer())
                .create();
    }

    public static Optional<State> loadFrom(File file) {
        initGsonIfNull();
        try (FileReader reader = new FileReader(file)) {
            return Optional.of(gson.fromJson(reader, State.class));
        } catch (Exception e) {
            return Optional.empty();
        }
    }

    public static void saveTo(File file, State state) throws IOException {
        initGsonIfNull();
        try (FileWriter writer = new FileWriter(file)) {
            gson.toJson(state, writer);
        }
    }

import java.io.IOException;
import java.util.List;

import com.googlecode.lanterna.TerminalSize;
import com.googlecode.lanterna.TextColor;
import com.googlecode.lanterna.graphics.TextGraphics;
import com.googlecode.lanterna.input.KeyStroke;
import com.googlecode.lanterna.input.KeyType;
import com.googlecode.lanterna.terminal.DefaultTerminalFactory;
import com.googlecode.lanterna.terminal.Terminal;

import app.fooddelivery.service.Restaurant;

public class Utils {
        public interface Screen {
                Screen run(Term term, Restaurant restaurant);
        }

        public interface Line {
                String render(boolean selected);

                void onKey(KeyStroke key);
        }

        interface Action {
                Screen run();
        }

        public static class Term {
                private Terminal terminal;
                private TextGraphics text;
                private TerminalSize size;

                public Term() {
                        DefaultTerminalFactory defaultTerminalFactory = new DefaultTerminalFactory();
                        try {
                                this.terminal = defaultTerminalFactory.createTerminal();
                                this.size = terminal.getTerminalSize();
                                text = terminal.newTextGraphics().setForegroundColor(TextColor.ANSI.GREEN);
                        } catch (IOException e) {
                                System.err.println("Failed to create terminal");
                        }
                }

                public Terminal terminal() {
                        return terminal;
                }

                public TextGraphics text() {
                        return text;
                }

                public TerminalSize size() {
                        return size;
                }

                public KeyStroke readKey() {
                        try {
                                return terminal.readInput();
                        } catch (IOException e) {
                                System.err.println("Failed to read terminal input");
                                return null;
                        }
                }

                public void flush() {
                        try {
                                terminal.flush();
                        } catch (IOException e) {
                                System.err.println("Failed to flush terminal");
                        }
                }

                public void clear() {
                        try {
                                terminal.clearScreen();
                        } catch (IOException e) {
                                System.err.println("Failed to clear terminal");

                        }
                }

        }

        public static class Text implements Line {
                private String label;

                Text(String label) {
                        this.label = label;
                }
                @Override
                public String render(boolean selected) {
                        return label;
                }
                @Override
                public void onKey(KeyStroke key) {
                }
        }

        public static class Button implements Line {
                private String label;
                private Action action;
                private Screen result = null;

                Button(String label, Action action) {
                        this.label = label;
                        this.action = action;
                }
                @Override
                public String render(boolean selected) {
                        return (selected ? "> " : " ") + label;
                }
                @Override
                public void onKey(KeyStroke key) {
                        if (key.getKeyType() == KeyType.Enter) {
                                result = action.run();
                        }
                }

                public Screen result() {
                        Screen temp = result;
                        result = null;
                        return temp;
                }
        }

        public static class Amount implements Line {
                private final String label;
                private int quantity;
                final int min;
                final int max;

                Amount(String label) {
                        this.label = label;
                        this.min = 0;
                        this.max = 100;
                        this.quantity = this.min;
                }

                Amount(String label, int min, int max) {
                        this.label = label;
                        this.quantity = min;
                        this.min = min;
                        this.max = max;
                }
                @Override
                public String render(boolean selected) {
                        return (selected ? "> " : " ") + label + " - " + quantity + " + ";
                }
                @Override
                public void onKey(KeyStroke key) {
                        if (key.getKeyType() == KeyType.ArrowRight && quantity < max) {
                                quantity++;
                        }
                        if (key.getKeyType() == KeyType.ArrowLeft && quantity > min) {
                                quantity--;
                        }
                }

                public int quantity() {
                        return quantity;
                }
        }

        public static class Input implements Line {
                private final String label;
                private String value = "";
                private final boolean hidden;

                Input(String label, boolean hidden) {
                        this.label = label;
                        this.hidden = hidden;
                }
                @Override
                public String render(boolean selected) {
                        String display = hidden ? "*".repeat(value.length()) : value;
                        return (selected ? "> " : " ") + label + " " + display;
                }
                @Override
                public void onKey(KeyStroke key) {
                        if (key.getKeyType() == KeyType.Character) {
                                value += key.getCharacter();
                        }
                        if (key.getKeyType() == KeyType.Backspace && value.length() > 0) {
                                value = value.substring(0, value.length() - 1);
                        }
                }

                public String getInput() {
                        return value;
                }
        }
        
        public static void render(List<Line> lines, int selection, TerminalSize screenSize, TextGraphics text, int y) {
                for (int i = 0; i < lines.size(); i++) {
                        String line = lines.get(i).render(i == selection);

                        text.putString(0, y + i, " ".repeat(screenSize.getColumns()));
                        int x = screenSize.getColumns() / 2 - line.length() / 2;
                        text.putString(x, y + i, line);
                }
        }
        public static void render(Line line, int selection, TerminalSize screenSize, TextGraphics text, int y) {
                        String txt = line.render(false);
                        text.putString(0, y, " ".repeat(screenSize.getColumns()));
                        int x = screenSize.getColumns() / 2 - txt.length() / 2;
                        text.putString(x, y, txt);
        }


        public static int keyPress(int selection, KeyStroke key, List<Line> lines) {
                if (key.getKeyType() == KeyType.ArrowDown)
                        selection++;
                else if (key.getKeyType() == KeyType.ArrowUp)
                        selection--;
                else if (key.getKeyType() == KeyType.Enter && (lines.get(selection) instanceof Utils.Input
                                || lines.get(selection) instanceof Utils.Amount))
                        selection++;

                if (selection < 0)
                        selection = lines.size() - 1;
                if (selection >= lines.size())
                        selection = 0;

                return selection;
        }

        public static Screen loop(List<Line> title, List<Line> lines, Term term) {
                int selection = 0;

                render(title, -1, term.size(), term.text(), 0);
                render(lines, selection, term.size(), term.text(), 3);
                Line footer = new Text("↑↓ Navigate   Enter Select   ←→ Change Quantity");
                render(footer, -1, term.size(), term.text(), term.size().getRows() - 1);

                term.flush();

                while (true) {
                        KeyStroke key = term.readKey();
                        if (key == null) {
                                continue;
                        }
                        Line current = lines.get(selection);
                        current.onKey(key);

                        if (current instanceof Button) {
                                Button b = (Button) current;
                                Screen next = b.result();
                                if (next != null) {
                                        return next;
                                }
                        }

                        selection = keyPress(selection, key, lines);

                        render(title, -1, term.size(), term.text(), 0);
                        render(lines, selection, term.size(), term.text(), 3);
                        render(footer, -1, term.size(), term.text(), term.size().getRows() - 1);


                        term.flush();
                }
        }

        public static Screen loop(List<Line> title, List<Line> lines, Term term, int y) {
                int selection = 0;

                render(title, -1, term.size(), term.text(), 0);
                render(lines, selection, term.size(), term.text(), y);
                Line footer = new Text("↑↓ Navigate   Enter Select   ←→ Change Quantity");
                render(footer, -1, term.size(), term.text(), term.size().getRows() - 1);

                term.flush();

                while (true) {
                        KeyStroke key = term.readKey();
                        
                        if (key == null) {
                                continue;
                        }

                        Line current = lines.get(selection);
                        current.onKey(key);

                        if (current instanceof Button) {
                                Button b = (Button) current;
                                Screen next = b.result();
                                if (next != null) {
                                        return next;
                                }
                        }

                        selection = keyPress(selection, key, lines);

                        render(title, -1, term.size(), term.text(), 0);
                        render(lines, selection, term.size(), term.text(), y);
                        render(footer, -1, term.size(), term.text(), term.size().getRows() - 1);

                        term.flush();
                }
        }
